    <script>

        // --- ZDE UPRAV ODKAZY NA FOTKY NA FTP ---

        // Nahraď testovací odkazy za cesty k tvým fotkám (např. "fotky/1.jpg")

        const photos = [

            "fotky“/1.jpg“,

            "fotky“/2.jpg“,

            "fotky“/3.jpg“

        ];

        

        let currentPhotoIndex = 0;

        const container = document.getElementById('photo-container');

        const photoEl = document.getElementById('photo');

        const btn = document.getElementById('more-love-btn');


        // Funkce pro přidání polibku

        function addKiss(e) {

            e.preventDefault(); // Důležité pro mobily (zabrání posunu obrazovky)


            // Získání správných souřadnic (pro dotyk i myš)

            let clientX, clientY;

            if (e.touches && e.touches.length > 0) {

                clientX = e.touches[0].clientX;

                clientY = e.touches[0].clientY;

            } else {

                clientX = e.clientX;

                clientY = e.clientY;

            }


            // Výpočet pozice přesně v rámci fotky

            const rect = container.getBoundingClientRect();

            const x = clientX - rect.left;

            const y = clientY - rect.top;


            // Vytvoření HTML elementu pro polibek

            const kiss = document.createElement('div');

            kiss.classList.add('kiss');

            kiss.innerText = '💋';

            

            // Nastavení umístění (souřadnice X a Y)

            kiss.style.left = `${x}px`;

            kiss.style.top = `${y}px`;


            // Náhodné otočení od 0 do 360 stupňů

            const randomRotation = Math.floor(Math.random() * 360);

            kiss.style.setProperty('--rot', `${randomRotation}deg`);


            // Vložení polibku do kontejneru k fotce

            container.appendChild(kiss);

        }


        // Funkce pro načtení další fotky

        function nextPhoto() {

            // Smazání všech stávajících polibků

            const kisses = container.querySelectorAll('.kiss');

            kisses.forEach(k => k.remove());


            // Posun na další fotku (pokud jsme na konci, vrátí se na začátek)

            currentPhotoIndex = (currentPhotoIndex + 1) % photos.length;

            photoEl.src = photos[currentPhotoIndex];

        }


        // Přiřazení akcí (kliknutí / dotyk)

        container.addEventListener('pointerdown', addKiss);

        btn.addEventListener('click', nextPhoto);
